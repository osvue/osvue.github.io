# 自定义节点

<!--
工作流节点组件 (WorkflowNode.vue)
功能：自定义工作流节点组件，支持状态显示、执行次数统计、详情查看等功能

使用方式：
1. 基础使用：
   <WorkflowNode
     :data="nodeData"
     :selected="false"
     @click="handleClick"
     @view-details="handleViewDetails"
   />

2. 节点数据结构：
   {
     id: string | number,      // 节点唯一标识
     label: string,            // 节点标签
     status: string,           // 节点状态
     executionCount: number,   // 执行次数
     executions: Array         // 执行记录
   }

3. 状态类型：
   - completed: 已完成（绿色）
   - in-progress: 进行中（蓝色）
   - pending: 待执行（黄色）
   - failed: 失败（红色）

扩展方式：
1. 添加新的状态类型：
   - 在 getStatusType 和 getStatusText 中添加新的状态处理
   - 在样式类中添加对应的状态样式

2. 自定义节点内容：
   - 修改模板结构
   - 添加新的数据展示
   - 添加新的交互按钮

3. 自定义样式：
   - 修改 .workflow-node 相关样式
   - 添加新的状态样式类
   - 自定义动画效果

4. 添加新的交互功能：
   - 添加新的事件处理函数
   - 通过 emit 向父组件传递事件
-->

<template>
  <div
    class="workflow-node"
    :class="{
      'node-completed': data.status === 'completed',
      'node-in-progress': data.status === 'in-progress',
      'node-pending': data.status === 'pending',
      'node-failed': data.status === 'failed',
      'node-selected': selected
    }"
    @click="$emit('click', $event)"
  >
    <!-- 节点头部：标题和状态 -->
    <div class="node-header">
      <span class="node-title">{{ data.label }}</span>
      <el-tag
        :type="getStatusType(data.status)"
        size="small"
        class="node-status-tag"
      >
        {{ getStatusText(data.status) }}
      </el-tag>
    </div>
    <!-- 节点内容：执行次数和操作按钮 -->
    <div class="node-content">
      <div class="execution-count">
        <el-icon><Timer /></el-icon>
        <span>执行次数: {{ data.executionCount }}</span>
      </div>
      <div class="node-actions">
        <el-button
          type="primary"
          size="small"
          :icon="View"
          circle
          @click.stop="handleViewDetails"
        />
      </div>
    </div>
  </div>
</template>

<script setup>
import { Timer, View } from '@element-plus/icons-vue'

// 组件属性定义
const props = defineProps({
  data: {
    type: Object,
    required: true,
    // 数据结构验证
    validator: (value) => {
      return value.id !== undefined && 
             value.label !== undefined && 
             value.status !== undefined
    }
  },
  selected: {
    type: Boolean,
    default: false
  }
})

// 定义组件事件
const emit = defineEmits(['click', 'view-details'])

/**
 * 处理查看详情按钮点击
 * 向父组件传递完整的节点数据
 */
const handleViewDetails = () => {
  emit('view-details', {
    id: props.data.id,
    title: props.data.label,
    status: props.data.status,
    executionCount: props.data.executionCount,
    executions: props.data.executions
  })
}

/**
 * 获取状态对应的标签类型
 * @param {string} status - 节点状态
 * @returns {string} Element Plus 标签类型
 * 
 * 扩展说明：
 * 1. 可以添加新的状态类型
 * 2. 可以自定义状态颜色
 * 3. 可以添加状态图标
 */
const getStatusType = (status) => {
  switch (status) {
    case 'completed':
      return 'success'  // 绿色
    case 'in-progress':
      return 'primary'  // 蓝色
    case 'pending':
      return 'info'     // 灰色
    case 'failed':
      return 'danger'   // 红色
    default:
      return 'info'     // 默认灰色
  }
}

/**
 * 获取状态对应的文本
 * @param {string} status - 节点状态
 * @returns {string} 状态文本
 * 
 * 扩展说明：
 * 1. 可以添加新的状态文本
 * 2. 可以支持国际化
 * 3. 可以添加状态描述
 */
const getStatusText = (status) => {
  switch (status) {
    case 'completed':
      return '已完成'
    case 'in-progress':
      return '进行中'
    case 'pending':
      return '待执行'
    case 'failed':
      return '失败'
    default:
      return '未知'
  }
}
</script>

<!--
工作流程图组件 (WorkflowGraph.vue)
功能：基于 vue-flow 实现的可交互工作流程图，支持节点拖拽、缩放、连线等功能

使用方式：

   <WorkflowGraph 
     :nodes="workflowNodes"
     :default-viewport="{ x: 0, y: 0, zoom: 1 }"
     :min-zoom="0.2"
     :max-zoom="4"
     :snap-to-grid="true"
     :snap-grid="[15, 15]"
     @node-click="handleNodeClick"
     @pane-click="onPaneClick"
   />

Props 说明：
- nodes: Array<WorkflowNode> - 工作流节点数据
  WorkflowNode 结构：
  {
    id: string | number,      // 节点唯一标识
    title: string,            // 节点标题
    type: string,             // 节点类型（可选，用于自定义节点样式）
    status: string,           // 节点状态（completed/in-progress/pending/failed）

  }

Events 说明：
- node-click: (nodeData: WorkflowNode) => void - 节点点击事件
- pane-click: (event: MouseEvent) => void - 画布点击事件

扩展方式：
1. 自定义节点样式：
   - 修改 WorkflowNode.vue 中的样式
   - 添加新的节点类型和对应的样式类

2. 自定义连线样式：
   - 修改 transformNodesToElements 中的连线配置
   - 添加新的连线类型和样式

3. 添加交互功能：
   - 在 template 中添加新的 vue-flow 事件监听
   - 实现对应的事件处理函数
   - 通过 emit 向父组件传递事件

4. 添加工具栏：
   - 在 Controls 组件中添加自定义按钮

-->

<template>

<script setup>
import { ref, computed, watch, onMounted } from 'vue'
import { VueFlow, useVueFlow } from '@vue-flow/core'
import { Background } from '@vue-flow/background'
import { MiniMap } from '@vue-flow/minimap'
import { Controls } from '@vue-flow/controls'
import '@vue-flow/core/dist/style.css'
import '@vue-flow/core/dist/theme-default.css'
import '@vue-flow/minimap/dist/style.css'
import '@vue-flow/controls/dist/style.css'
import WorkflowNode from './components/WorkflowNode.vue'

// 定义组件事件
const emit = defineEmits(['node-click'])

// 组件属性定义
const props = defineProps({
  nodes: {
    type: Array,
    required: true,
    // 节点数据结构验证
    validator: (value) => {
      return value.every(node => 
        node.id !== undefined && 
        node.title !== undefined &&
        node.status !== undefined
      )
    }
  },
})

/**
 * 计算节点位置
 * @param {number} index - 节点索引
 * @param {number} totalNodes - 总节点数
 * @returns {{x: number, y: number}} 节点坐标
 * 
 * 扩展说明：
 * 1. 可以修改此函数实现不同的布局算法
 * 2. 可以添加参数支持自定义布局参数
 * 3. 可以实现自动布局算法（如力导向布局）
 */
const calculateNodePosition = (index, totalNodes) => {
  const padding = 100
  const nodeWidth = 200
  const nodeHeight = 100
  const horizontalSpacing = 300
  const verticalSpacing = 150

  // 计算节点位置
  const x = padding + (index * horizontalSpacing)
  const y = padding + (Math.floor(index / 2) * verticalSpacing)

  return { x, y }
}

/**
 * 转换节点数据为 vue-flow 格式
 * @param {Array} nodes - 原始节点数据
 * @returns {Array} vue-flow 元素数组
 * 
 * 扩展说明：
 * 1. 可以添加更多节点类型
 * 2. 可以自定义连线样式
 * 3. 可以实现节点分组
 * 4. 可以添加节点标签
 */
const transformNodesToElements = (nodes) => {
  if (!nodes || nodes.length === 0) return []

  const elements = []
  const totalNodes = nodes.length

  // 添加节点
  nodes.forEach((node, index) => {
    const position = calculateNodePosition(index, totalNodes)
    
    elements.push({
      id: node.id.toString(),
      type: 'custom', // 使用自定义节点类型
      position,
      data: {
        ...node,
        label: node.title,
        status: node.status,
        executionCount: node.executionCount
    }
  })

    // 添加连接线
    if (index < totalNodes - 1) {
      elements.push({
        id: `edge-${node.id}-${nodes[index + 1].id}`,
        source: node.id.toString(),
        target: nodes[index + 1].id.toString(),
        type: 'smoothstep', // 平滑的连线样式
        animated: node.status === 'in-progress', // 进行中的节点连线动画
        style: { stroke: getConnectionColor(node.status, nodes[index + 1].status) }
      })
    }
  })

  return elements
}

/**
 * 获取连接线颜色
 * @param {string} fromStatus - 起始节点状态
 * @param {string} toStatus - 目标节点状态
 * @returns {string} 颜色代码
 * 
 * 扩展说明：
 * 1. 可以添加更多状态对应的颜色
 * 2. 可以实现渐变色
 * 3. 可以添加自定义状态
 */
const getConnectionColor = (fromStatus, toStatus) => {
  if (fromStatus === 'completed' && toStatus === 'completed') {
    return '#67c23a' // 完成状态 - 绿色
  } else if (fromStatus === 'completed' && toStatus === 'in-progress') {
    return '#67c23a' // 完成到进行中 - 绿色
  } else if (fromStatus === 'in-progress') {
    return '#409eff' // 进行中 - 蓝色
  } else {
    return '#909399' // 其他状态 - 灰色
  }
}
