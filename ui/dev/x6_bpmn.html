// 在 beforeEnter 钩子中根据某些条件进行重定向
  const routess = {
    path: '/',
    name: 'layout',
    component: () => import('@/components/layout/layout.vue'),
    beforeEnter: (to, from, next) => {
      // 基于某些条件来判断重定向目标
      const shouldRedirect = true; // 你的条件
      if (shouldRedirect) {
        next('/task');
      } else {
        next();
      }
    },
 }
