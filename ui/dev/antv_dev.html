# antv tree

<template>
  <a-tree
    :treeData="treeData"
    :fieldNames="fieldNames"
    :checkable="true"
    :checkedKeys="checkedKeys"
    @check="onCheck"
    :expandedKeys="expandedKeys"
    @expand="onExpand"
    defaultExpandAll
    :showLine="true"
    :selectable="false"
  >
    <template #title="{dataRef  }">
      <!-- 根据节点类型渲染不同的表单元素 -->
      <div>
        <!-- {{ dataRef }} -->
         <span v-if="dataRef.type === 'input'">
          <a-input
            v-model:value="dataRef.value"
            :placeholder="dataRef.placeholder || '请输入内容'"
          />
        </span>
        <span v-if="dataRef.type === 'select'">
          <a-select v-model:value="dataRef.value" :placeholder="dataRef.placeholder || '请选择'">
            <a-select-option v-for="option in dataRef.options" :key="option" :value="option">
              {{ option }}
            </a-select-option>
          </a-select>
        </span>
        <span v-if="dataRef.type === 'checkbox'">
          <a-checkbox v-model:checked="dataRef.value">{{ dataRef.label }}</a-checkbox>
        </span>
        <span v-if="dataRef.type === 'radio'">
          <a-radio-group v-model:value="dataRef.value">
            <a-radio v-for="option in dataRef.options" :key="option" :value="option">
              {{ option }}
            </a-radio>
          </a-radio-group>
        </span> 
         <span v-if="!dataRef.type">
          {{ dataRef.title }}  
        </span>  
      </div>
    </template>
  </a-tree>

  <!-- 全选/取消全选按钮 -->
  <a-button @click="toggleSelectAll">全选/取消全选</a-button>
  <!-- 删除选中节点按钮 -->
  <a-button @click="deleteSelectedNodes">删除选中节点</a-button>
</template>

<script setup>
import { ref } from 'vue'
import { Tree, Button, Checkbox, Input, Select, Radio } from 'ant-design-vue'

const treeData = ref([
  {
    title: '父节点 1',
    key: '0-0',
    children: [
      {
        title: '子节点 1-1',
        key: '0-0-0',
        selected: false,
        type: 'input', // 类型为输入框
        value: '',
        placeholder: '请输入文本',
        showCheckbox: true,
        isNew: false,
        children: []
      },
      {
        title: '子节点 1-2',
        key: '0-0-1',
        selected: false,
        type: 'select', // 类型为下拉框
        value: '',
        placeholder: '请选择选项',
        options: ['选项1', '选项2', '选项3'],
        showCheckbox: false,
        isNew: false,
        children: []
      }
    ]
  },
  {
    title: '父节点 2',
    key: '0-1',
    children: [
      {
        title: '子节点 2-1',
        key: '0-1-0',
        selected: false,
        type: 'checkbox', // 类型为复选框
        value: false,
        label: '是否启用',
        showCheckbox: true,
        isNew: true,
        children: []
      },
      {
        title: '子节点 2-2',
        key: '0-1-1',
        selected: false,
        type: 'radio', // 类型为单选框
        value: '',
        options: ['是', '否'],
        showCheckbox: true,
        isNew: true,
        children: []
      }
    ]
  }
])

// 自定义字段映射
const fieldNames = {
  title: 'title',
  key: 'key',
  children: 'children'
}

// 选中的节点的 keys
const checkedKeys = ref([])
// 展开的节点的 keys
const expandedKeys = ref([])

// 处理节点的选中/取消选中
const onCheck = (checkedKeysValue) => {
  checkedKeys.value = checkedKeysValue
}

// 处理节点的展开/折叠
const onExpand = (expandedKeysValue) => {
  expandedKeys.value = expandedKeysValue
}

// 全选/取消全选
const toggleSelectAll = () => {
  const allKeys = getAllKeys(treeData.value)
  if (checkedKeys.value.length === allKeys.length) {
    checkedKeys.value = [] // 取消全选
  } else {
    checkedKeys.value = allKeys // 全选
  }
}

// 获取所有节点的 key
const getAllKeys = (nodes) => {
  let keys = []
  nodes.forEach(node => {
    keys.push(node.key)
    if (node.children && node.children.length > 0) {
      keys = keys.concat(getAllKeys(node.children))
    }
  })
  return keys
}

// 删除选中的节点
const deleteSelectedNodes = () => {
  const selectedKeys = checkedKeys.value
  deleteNodes(treeData.value, selectedKeys)
}

// 递归删除节点
const deleteNodes = (nodes, selectedKeys) => {
  for (let i = 0; i < nodes.length; i++) {
    const node = nodes[i]
    if (selectedKeys.includes(node.key)) {
      nodes.splice(i, 1) // 删除该节点
      i-- // 删除后需要回退索引
    } else if (node.children && node.children.length > 0) {
      deleteNodes(node.children, selectedKeys)
    }
  }
}
</script>

# 原生html tree

<template>
  <div>
    <!-- 全选 / 取消全选 -->
    <button @click="toggleSelectAll">全选/取消全选</button>

    <div v-for="(node, index) in treeData" :key="node.id" class="tree-node">
      <div class="node-label">
        <!-- 多选框 (仅在 isNew 为 true 时显示) -->
        <input
          type="checkbox"
          v-model="node.selected"
          v-if="node.isNew"
          @change="handleCheckboxChange(node)"
          @click.stop="handleCheckboxClick(node, index, $event)"
        />

        <span @click="toggleCollapse(node)">
          {{ node.label }} (porder: {{ node.porder }})
        </span>

        <!-- 根据节点类型渲染不同的控件 -->
        <template v-if="node.type === 'input'">
          <input v-model="node.value" :placeholder="node.placeholder" />
        </template>
        <template v-else-if="node.type === 'select'">
          <select v-model="node.value">
            <option v-for="option in node.options" :key="option" :value="option">
              {{ option }}
            </option>
          </select>
        </template>
        <template v-else-if="node.type === 'checkbox'">
          <input type="checkbox" v-model="node.value" />
        </template>
        <template v-else>
          <input type="text" v-model="node.value" />
        </template>

        <!-- 折叠按钮 -->
        <button @click="toggleCollapse(node)">
          {{ node.collapsed ? "展开" : "折叠" }}
        </button>

        <!-- 添加子节点按钮 -->
        <button @click="addChildNode(node)">添加子节点</button>
        <!-- 添加同级节点按钮 -->
        <button @click="addSiblingNode(node, index)">添加同级节点</button>
        <!-- 删除节点按钮 -->
        <button @click="deleteNode(node, index)">删除节点</button>
      </div>

      <!-- 递归渲染子节点 -->
      <div v-if="node.children && node.children.length && !node.collapsed">
        <Tree :treeData="node.children" />
      </div>
    </div>
  </div>
</template>

<script setup>
import { defineProps, ref } from "vue";

const props = defineProps({
  treeData: {
    type: Array,
    required: true,
  },
});

// 记录最后选中的索引
const lastSelectedIndex = ref(null);

// 递归折叠/展开节点
function toggleCollapse(node) {
  node.collapsed = !node.collapsed;
}

// ✅ 选择某个节点时，所有子节点也同步选中
function handleCheckboxChange(node) {
  setChildrenSelected(node, node.selected);
}

// 递归设置所有子节点选中状态
function setChildrenSelected(node, selected) {
  if (node.children && node.children.length > 0) {
    node.children.forEach((child) => {
      child.selected = selected;
      setChildrenSelected(child, selected); // 递归处理子节点
    });
  }
}

// ✅ 按住 Shift 选择多个节点
function handleCheckboxClick(node, index, event) {
  if (event.shiftKey && lastSelectedIndex.value !== null) {
    const start = Math.min(lastSelectedIndex.value, index);
    const end = Math.max(lastSelectedIndex.value, index);
    for (let i = start; i <= end; i++) {
      props.treeData[i].selected = true;
    }
  } else {
    node.selected = !node.selected;
    setChildrenSelected(node, node.selected); // 确保子节点状态同步
  }

  lastSelectedIndex.value = index;
}

// ✅ 全选 / 取消全选
function toggleSelectAll() {
  const allSelected = props.treeData.every((node) => node.selected);
  props.treeData.forEach((node) => {
    node.selected = !allSelected;
    setChildrenSelected(node, !allSelected); // 级联选中所有子节点
  });
}

// ✅ 递归删除节点，并重新排序 porder
function deleteNode(node, index) {
  if (node.selected) {
    if (index >= 0) {
      props.treeData.splice(index, 1);
      updatePorder(props.treeData); // 重新排序 porder
    } else {
      const parentNode = findParentNode(props.treeData, node);
      if (parentNode) {
        const childIndex = parentNode.children.indexOf(node);
        parentNode.children.splice(childIndex, 1);
        updatePorder(parentNode.children); // 重新排序父节点下的 porder
      }
    }
  }
}

// 查找节点的父节点
function findParentNode(nodes, targetNode) {
  for (let node of nodes) {
    if (node.children && node.children.includes(targetNode)) {
      return node;
    }
    if (node.children && node.children.length > 0) {
      const parent = findParentNode(node.children, targetNode);
      if (parent) return parent;
    }
  }
  return null;
}

// ✅ 重新计算 porder（保证所有兄弟节点的 porder 连续）
function updatePorder(nodes) {
  nodes.forEach((node, index) => {
    node.porder = index;
  });
}

// 添加子节点
function addChildNode(node) {
  const newNode = {
    id: Date.now(),
    label: "新节点",
    type: "input",
    value: "",
    placeholder: "请输入内容",
    collapsed: false,
    children: [],
    selected: false,
    isNew: true,
    porder: node.children ? node.children.length : 0, // 子节点 porder 递增
  };
  if (!node.children) {
    node.children = [];
  }
  node.children.push(newNode);


  const newNode1 = {
    id: Date.now(),
    label: "新节点",
    type: "input",
    value: "",
    placeholder: "请输入内容",
    collapsed: false,
    children: [],
    selected: false,
    isNew: true,
    porder: node.children ? node.children.length : 0, // 子节点 porder 递增
  };
 
  node.children.push(newNode1);

}

// 添加同级节点
function addSiblingNode(node, index) {
  const newNode = {
    id: Date.now(),
    label: "新同级节点",
    type: "input",
    value: "",
    placeholder: "请输入内容",
    collapsed: false,
    children: [],
    selected: false,
    isNew: true,
  };
  props.treeData.splice(index + 1, 0, newNode);
  updatePorder(props.treeData); // 重新排序同级节点的 porder
}
</script>
