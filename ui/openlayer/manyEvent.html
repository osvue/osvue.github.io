# OpenLayers 复杂图层点击事件处理指南

# 1. 图层结构分析

# 2. 点击事件处理流程

# 2.1 事件监听设置

// 在地图初始化时添加点击事件监听
glMap.on('click', (event) => {
  const pixel = event.pixel;        // 获取点击像素坐标
  const coordinate = event.coordinate; // 获取点击地理坐标
  
  // 存储点击位置的所有要素
  const clickedFeatures = [];
  
  // 遍历点击位置的所有图层
  glMap.forEachFeatureAtPixel(pixel, (feature, layer) => {
    clickedFeatures.push({
      feature: feature,
      layer: layer,
      layerName: layer.get('name')
    });
  });
});

# 2.2 图层优先级处理

// 定义图层优先级顺序
const priorityOrder = ['markerLayer', 'basicInfoLayer', 'farmBoundary'];

// 按优先级排序点击到的要素
clickedFeatures.sort((a, b) => {
  return priorityOrder.indexOf(a.layerName) - priorityOrder.indexOf(b.layerName);
});

// 处理最高优先级的图层点击事件
const topFeature = clickedFeatures[0];
const layerName = topFeature.layerName;
const feature = topFeature.feature;

# 7. 示例代码

// 图层配置常量
const LAYER_NAMES = {
  MARKER: 'markerLayer',
  BASIC_INFO: 'basicInfoLayer',
  FARM_BOUNDARY: 'farmBoundary'
};

// 图层优先级配置
const LAYER_PRIORITY = [
  LAYER_NAMES.MARKER,
  LAYER_NAMES.BASIC_INFO,
  LAYER_NAMES.FARM_BOUNDARY
];

// 点击事件处理函数
function handleMapClick(event) {
  const { pixel, coordinate } = event;
  const clickedFeatures = [];
  
  // 收集点击位置的要素
  glMap.forEachFeatureAtPixel(pixel, (feature, layer) => {
    clickedFeatures.push({
      feature,
      layer,
      layerName: layer.get('name')
    });
  });
  
  // 按优先级处理
  if (clickedFeatures.length > 0) {
    const topFeature = clickedFeatures.sort((a, b) => {
      return LAYER_PRIORITY.indexOf(a.layerName) - 
             LAYER_PRIORITY.indexOf(b.layerName);
    })[0];
    
    // 处理点击事件
    handleFeatureClick(topFeature, coordinate);
  } else {
    // 点击空白处
    clearSelection();
  }
}

// 要素点击处理
function handleFeatureClick({ feature, layerName }, coordinate) {
  try {
    switch (layerName) {
      case LAYER_NAMES.MARKER:
        handleMarkerClick(feature, coordinate);
        break;
      case LAYER_NAMES.BASIC_INFO:
        handleBasicInfoClick(feature, coordinate);
        break;
      case LAYER_NAMES.FARM_BOUNDARY:
        handleFarmBoundaryClick(feature, coordinate);
        break;
    }
  } catch (error) {
    handleClickError(error);
  }
}
