  1. 图层源（Source）
// 图层源是数据的来源，定义了如何获取和存储地理数据
// 常见的图层源类型：

// 1. VectorSource - 矢量数据源
const vectorSource = new VectorSource({
  features: features,  // 直接提供要素数组
  // 或者通过 URL 加载数据
  url: 'path/to/geojson',
  format: new GeoJSON()
});

// 2. TileSource - 瓦片数据源
const tileSource = new XYZ({
  url: 'http://example.com/tiles/{z}/{x}/{y}.png'
});

// 3. ImageSource - 图片数据源
const imageSource = new ImageWMS({
  url: 'http://example.com/geoserver/wms',
  params: {
    LAYERS: 'layer_name',
    FORMAT: 'image/png'
  }
});
  1. 图层（Layer）
// 图层是数据的可视化表现，定义了如何渲染数据
// 常见的图层类型：

// 1. VectorLayer - 矢量图层
const vectorLayer = new VectorLayer({
  name: 'myVectorLayer',  // 图层名称，用于识别
  source: vectorSource,   // 关联的图层源
  style: new Style({      // 样式定义
    fill: new Fill({ color: 'rgba(255,0,0,0.1)' }),
    stroke: new Stroke({ color: 'red', width: 2 })
  }),
  zIndex: 1,             // 图层层级
  visible: true,         // 是否可见
  opacity: 0.8          // 透明度
});

// 2. TileLayer - 瓦片图层
const tileLayer = new TileLayer({
  source: tileSource,
  zIndex: 0
});

// 3. ImageLayer - 图片图层
const imageLayer = new ImageLayer({
  source: imageSource,
  zIndex: 2
});
  1. 图层组（LayerGroup）
// 图层组用于管理多个图层，可以统一控制多个图层的显示状态
const layerGroup = new LayerGroup({
  name: 'myLayerGroup',
  layers: [
    vectorLayer,
    tileLayer,
    imageLayer
  ],
  // 图层组可以设置统一的属性
  visible: true,
  opacity: 0.8,
  // 可以设置图层组的样式
  style: new Style({
    fill: new Fill({ color: 'rgba(0,0,255,0.1)' })
  })
});
  1. 图层层级和显示效果管理：
// 1. 设置图层层级
// 方法一：创建时设置
const layer1 = new VectorLayer({
  zIndex: 1,
  // ... 其他配置
});

// 方法二：动态设置
layer1.setZIndex(2);

// 2. 管理图层组
// 创建多个图层组
const baseGroup = new LayerGroup({
  name: 'baseGroup',
  layers: [tileLayer],
  zIndex: 0
});

const overlayGroup = new LayerGroup({
  name: 'overlayGroup',
  layers: [vectorLayer, imageLayer],
  zIndex: 1
});

// 3. 图层组嵌套
const mainGroup = new LayerGroup({
  name: 'mainGroup',
  layers: [baseGroup, overlayGroup],
  zIndex: 0
});

// 4. 控制图层显示
// 控制单个图层
layer1.setVisible(false);
layer1.setOpacity(0.5);

// 控制图层组
layerGroup.setVisible(false);
layerGroup.setOpacity(0.5);

// 5. 获取和遍历图层
// 获取地图上的所有图层
const allLayers = map.getLayers().getArray();

// 遍历图层组中的图层
layerGroup.getLayers().forEach(layer => {
  console.log(layer.get('name'));
  // 设置每个图层的属性
  layer.setZIndex(layer.getZIndex() + 1);
});

// 6. 图层排序
// 按 zIndex 排序图层
const sortedLayers = allLayers.sort((a, b) => {
  return a.getZIndex() - b.getZIndex();
});
  1. 实际应用示例：
// 创建一个包含多个图层组的地图
const map = new Map({
  target: 'map',
  layers: [
    // 底图组
    new LayerGroup({
      name: 'baseLayers',
      layers: [
        new TileLayer({
          name: 'satellite',
          source: new XYZ({
            url: 'http://example.com/satellite/{z}/{x}/{y}.png'
          }),
          zIndex: 0
        }),
        new TileLayer({
          name: 'terrain',
          source: new XYZ({
            url: 'http://example.com/terrain/{z}/{x}/{y}.png'
          }),
          zIndex: 1
        })
      ]
    }),
    // 数据图层组
    new LayerGroup({
      name: 'dataLayers',
      layers: [
        new VectorLayer({
          name: 'boundaries',
          source: new VectorSource({
            url: 'path/to/boundaries.geojson',
            format: new GeoJSON()
          }),
          zIndex: 2
        }),
        new VectorLayer({
          name: 'points',
          source: new VectorSource({
            url: 'path/to/points.geojson',
            format: new GeoJSON()
          }),
          zIndex: 3
        })
      ]
    })
  ]
});

// 控制图层组显示
function toggleLayerGroup(groupName, visible) {
  map.getLayers().forEach(layer => {
    if (layer instanceof LayerGroup && layer.get('name') === groupName) {
      layer.setVisible(visible);
    }
  });
}

// 调整图层顺序
function reorderLayers() {
  const layers = map.getLayers().getArray();
  layers.forEach((layer, index) => {
    layer.setZIndex(index);
  });
}

使用建议：

  1. 合理组织图层结构，使用图层组管理相关图层
  2. 为图层和图层组设置有意义的名称，便于管理
  3. 使用 zIndex 控制图层显示顺序
  4. 通过图层组统一管理多个图层的显示状态
  5. 注意图层性能，合理使用图层缓存和可见性控制
