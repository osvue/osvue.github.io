// 1. 图层定义和优先级设置
farmLayer = new VectorLayer({
  id: FARM_LAYER_ID,
  zIndex: 1,  // 较低的 zIndex，确保在底层
  source: vectorSource,
  // ... 样式配置
});

exampleLayer = new VectorLayer({
  id: EXAMPLE_LAYER_ID,
  zIndex: 2,  // 较高的 zIndex，确保在上层
  source: vectorSource,
  // ... 样式配置
});

// 2. 选择交互配置
const selectExampleClick = new Select({
  condition: click,  // 触发条件：点击
  style: selectStyle,
  layers: [exampleLayer],  // 只监听示例图层
  filter: function(feature, layer) {
    return layer.get('id') === EXAMPLE_LAYER_ID;  // 只处理示例图层的要素
  },
  hitTolerance: 5,    // 点击容差范围，使点击更容易触发
  multi: false,       // 禁用多选，一次只能选择一个要素
  stopClick: true     // 关键配置：阻止事件冒泡，防止事件传递到下层图层
});

const selectFarmClick = new Select({
  condition: click,
  style: selectStyle,
  layers: [farmLayer],  // 只监听农场图层
  filter: function(feature, layer) {
    return layer.get('id') === FARM_LAYER_ID;  // 只处理农场图层的要素
  },
  hitTolerance: 5,
  multi: false,
  stopClick: true
});

// 3. 交互添加顺序
glMap.addInteraction(selectExampleClick);  // 先添加示例图层交互
glMap.addInteraction(selectFarmClick);     // 后添加农场图层交互

// 4. 事件处理
selectExampleClick.on("select", function (e) {
  let features = e.target.getFeatures().getArray();
  if (features && features.length > 0) {
    // ... 处理示例图层点击
    selectFarmClick.getFeatures().clear();  // 清除农场图层的选择状态
  }
});

selectFarmClick.on("select", function (e) {
  let features = e.target.getFeatures().getArray();
  if (features && features.length > 0) {
    // ... 处理农场图层点击
    selectExampleClick.getFeatures().clear();  // 清除示例图层的选择状态
  }
});

点击事件没有被覆盖的关键原因：

  1. 图层隔离：

    • 每个选择交互通过 layers 属性指定了特定的图层
    • filter 函数进一步确保只处理对应图层的要素
    • 这样保证了两个图层的点击事件是相互独立的

  2. 事件冒泡控制：

    • stopClick: true 配置阻止了事件冒泡
    • 当点击上层图层时，事件不会传递到下层图层
    • 这确保了点击事件只会被一个图层处理

  3. 交互优先级：

    • 通过 zIndex 设置图层的视觉层级
    • 通过交互添加顺序控制事件处理优先级
    • 示例图层（zIndex: 2）的交互先添加，确保优先处理

  4. 选择状态管理：

    • 每个点击事件处理函数中都会清除另一个图层的选择状态
    • 这确保了同一时间只有一个图层处于选中状态
    • 避免了多个选择同时存在的情况

  5. 点击容差：

    • hitTolerance: 5 增加了点击的容差范围
    • 使点击更容易触发，提高了交互的准确性
    • 特别对于小区域的选择更有帮助

这种设计确保了：

  • 点击事件的处理是互斥的
  • 上层图层的点击不会被下层图层干扰
  • 每次点击只会触发一个图层的响应
  • 选择状态是清晰的，不会混淆

这就是为什么点击事件能够正确工作而不会被覆盖的原因。每个图层都有自己独立的交互处理机制，通过配置和状态管理确保了事件的正确触发和处理。
