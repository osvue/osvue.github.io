/**
 * 基础信息图层的样式函数
 * @param {Feature} feature - OpenLayers要素对象
 * @param {boolean} isHover - 是否为悬浮状态
 * @returns {Style} OpenLayers样式对象
 */
function basicInfoStyle(feature, isHover = false) {
  // 基础样式配置
  const baseStyle = {
    // 填充样式
    fill: new Fill({
      // 填充颜色，使用半透明的蓝色
      color: isHover ? 'rgba(0, 123, 255, 0.3)' : 'rgba(0, 123, 255, 0.1)'
    }),
    // 边框样式
    stroke: new Stroke({
      // 边框颜色，悬浮时加深
      color: isHover ? '#0056b3' : '#007bff',
      // 边框宽度，悬浮时加粗
      width: isHover ? 3 : 2,
      // 边框虚线样式，悬浮时使用实线
      lineDash: isHover ? [] : [5, 5]
    }),
    // 文本样式
    text: new Text({

/**
 * 创建点击事件处理器
 * @param {Map} map - OpenLayers地图实例
 * @param {Object} options - 配置选项
 * @returns {Function} 返回点击事件处理函数
 */
export function createClickHandler(map, options) {
  const {
    overlay,           // 弹窗覆盖物
    grazingDataArr,    // 数据数组
    globalClearLayer,  // 清除图层
    catPannel,         // 面板状态
    getFarmArea        // 获取农场面积函数
  } = options;

  // 创建一个变量来存储当前选中的图层
  let currentSelectedLayer = null;

  // 清除选中状态的函数
  function clearSelection() {
    // 清除之前选中的图层
    if (currentSelectedLayer) {
      currentSelectedLayer.getFeatures().clear();
      currentSelectedLayer = null;
    }
    // 隐藏面板
    catPannel.value = false;
    // 清除弹窗
    overlay.setPosition(undefined);
  }

  // 返回点击事件处理函数
  return function handleMapClick(event) {
    const { pixel, coordinate } = event;
    const clickedFeatures = [];

    // 收集点击位置的所有要素
    map.forEachFeatureAtPixel(pixel, (feature, layer) => {
      clickedFeatures.push({
        feature,
        layer,
        layerName: layer.get('name')
      });
    });

    if (clickedFeatures.length > 0) {
      // 按优先级排序处理点击事件
      clickedFeatures.sort((a, b) => {
        return LAYER_PRIORITY.indexOf(a.layerName) - 
               LAYER_PRIORITY.indexOf(b.layerName);
      });

      // 处理最高优先级的图层点击事件
      const topFeature = clickedFeatures[0];
      const { layerName, feature, layer } = topFeature;

      try {
        // 更新当前选中的图层
        currentSelectedLayer = layer;

        // 根据图层类型调用不同的处理函数
        switch (layerName) {
          case LAYER_NAMES.MARKER:
            handleMarkerClick(feature, coordinate);
            break;
          case LAYER_NAMES.BASIC_INFO:
            handleBasicInfoClick(feature, coordinate);
            break;
          case LAYER_NAMES.FARM_BOUNDARY:
            handleFarmBoundaryClick(feature, coordinate);
            break;
        }
      } catch (error) {
        console.error('点击事件处理错误:', error);
        clearSelection();
      }
    } else {
      // 点击空白处清除选中状态
      clearSelection();
    }
  };

  // 处理标记点点击
  function handleMarkerClick(feature, coordinate) {
    const markerName = feature.get('name');
    console.log('点击了标记点:', markerName);
    
    // 设置弹窗位置
    overlay.setPosition(coordinate);
    
    // 更新显示数据
    const markerData = [
      {
        key: "标记点名称",
        value: markerName
      },
      {
        key: "位置",
        value: `(${coordinate[0].toFixed(2)} E, ${coordinate[1].toFixed(2)} N)`
      }
    ];
    
    grazingDataArr.value = markerData;
  }

  // 处理基础信息点击
  function handleBasicInfoClick(feature, coordinate) {
    const farmName = feature.get('farmName');
    console.log('点击了基本信息图层:', farmName);
    
    overlay.setPosition(coordinate);
    
    const basicInfoData = [
      {
        key: "农场名称",
        value: farmName
      },
      {
        key: "点击位置",
        value: `(${coordinate[0].toFixed(2)} E, ${coordinate[1].toFixed(2)} N)`
      }
    ];
    
    grazingDataArr.value = basicInfoData;
  }

  // 处理农场边界点击
  function handleFarmBoundaryClick(feature, coordinate) {
    const farmBoundaryName = feature.get('name');
    console.log('点击了农场边界:', farmBoundaryName);
    
    overlay.setPosition(coordinate);
    const farmArea = getFarmArea(farmBoundaryName);
    
    const farmData = [
      {
        key: "试验点位置",
        value: farmBoundaryName
      },
      {
        key: "面积",
        value: `${farmArea} 亩`
      }
    ];
    
    grazingDataArr.value = farmData;
  }
}

import Feature from 'ol/Feature';
import Point from 'ol/geom/Point';
import { Icon, Style, Fill, Stroke, Circle, Text } from 'ol/style';

import { Vector as VectorLayer } from 'ol/layer';

/**
 * 创建自定义标记点图层
 * 该函数用于在地图上创建带有图标和标签的标记点图层
 * 
 * @param {Array} geometry - 经纬度坐标数组，格式：[[lng1, lat1], [lng2, lat2], ...]
 * @param {string} img - 图标路径或SVG字符串
 * @param {Array|string} label - 标签文本数组或单个文本
 * @param {Object} options - 配置选项
 * @param {Object} options.icon - 图标配置
 * @param {number} options.icon.size - 图标大小（像素）
 * @param {Array} options.icon.anchor - 图标锚点，默认为[0.5, 0.5]（图标中心点）
 * @param {Object} options.text - 文本配置
 * @param {string} options.text.font - 字体样式，例如：'16px Arial'
 * @param {string} options.text.color - 文本颜色，例如：'#fff'
 * @param {number} options.text.offsetY - 文本Y轴偏移量，负值向上偏移
 * @param {string} options.text.background - 文本背景色，例如：'rgba(0,0,0,0.6)'
 * @param {Array} options.text.padding - 文本内边距，例如：[2, 4]
 * @param {Object} options.hover - 悬停效果配置
 * @param {number} options.hover.scale - 悬停时图标缩放比例
 * @param {string} options.hover.color - 悬停时文本颜色
 * @returns {Object} 返回包含图层对象和相关方法的对象
 */
export function renderMakerCustom(
  geometry = [115.1, 30],
  img,
  label,
  options = {}
) {
  // 默认配置
  const defaultOptions = {
    icon: {
      size: 32,  // 调整图标大小为更合适的尺寸
      anchor: [0.5, 1]  // 修改锚点位置，使图标底部中心对齐坐标点
    },
    text: {
      font: 'bold 14px 微软雅黑',  // 使用更清晰的字体
      color: '#000',  // 使用黑色文字
      offsetY: -40,  // 增加文字与图标的距离
      background: 'rgba(255,255,255,0.8)',  // 使用白色半透明背景
      padding: [4, 8],  // 增加文字内边距
      stroke: {
        color: '#fff',  // 白色描边
        width: 4  // 增加描边宽度
      }
    },
    hover: {
      scale: 1.2,
      color: '#ff0000',  // 悬停时文字变红
      duration: 200
    }
  };

  // 合并用户配置和默认配置
  const config = {
    icon: { ...defaultOptions.icon, ...options.icon },
    text: { ...defaultOptions.text, ...options.text },
    hover: { ...defaultOptions.hover, ...options.hover }
  };

  // 创建样式缓存对象
  const styleCache = {
    normal: {},
    hover: {}
  };

  /**
   * 创建标记点样式
   * @param {Feature} feature - OpenLayers要素对象
   * @param {boolean} isHover - 是否为悬停状态
   * @returns {Style} OpenLayers样式对象
   */
  const createStyle = (feature, isHover = false) => {
    const name = feature.get('name');
    const cacheKey = `${name}_${isHover}`;
    
    if (styleCache[isHover ? 'hover' : 'normal'][cacheKey]) {
      return styleCache[isHover ? 'hover' : 'normal'][cacheKey];
    }

    const scale = isHover ? config.hover.scale : 1;
    const textColor = isHover ? config.hover.color : config.text.color;

    // 创建样式对象
    const style = new Style({
      // 图标样式
      image: new Icon({
        anchor: config.icon.anchor,
        src: "data:image/svg+xml," + encodeURIComponent(NetIcons.exchange),
        width: config.icon.size,
        height: config.icon.size,
         crossOrigin: 'anonymous'  // 添加跨域支持
      }),
      // 文本样式
      text: new Text({
        text: name,
        offsetY: config.text.offsetY,
        font: config.text.font,
        fill: new Fill({
          color: textColor
        }),
        stroke: new Stroke({
          color: config.text.stroke.color,
          width: config.text.stroke.width
        }),
        backgroundFill: new Fill({
          color: config.text.background
        }),
        padding: config.text.padding,
        textAlign: 'center',
        textBaseline: 'middle',
        offsetX: 0  // 确保文本水平居中
      })
    });

    // 缓存样式
    styleCache[isHover ? 'hover' : 'normal'][cacheKey] = style;
    return style;
  };

  // 创建标记点要素
  const markers = geometry.map((coord, index) => {
    const name = Array.isArray(label) ? label[index] : label;
    const feature = new Feature({
      geometry: new Point(coord),
      name: name,
      type: 'icon',
      properties: {
        index: index,
        originalCoord: coord
      }
    });
    
    // 直接设置样式，参考 renderMaker 的方式
    feature.setStyle(new Style({
      image: new Icon({
        anchor: [0.5, 0.5],
        src: "data:image/svg+xml," + encodeURIComponent(NetIcons.exchange),
        width: 50
      }),
      text: new Text({
        text: name,
        offsetY: -35,
        font: '20px Verdana,sans-serif',
        fill: new Fill({
          color: '#fff'
        })
      })
    }));
    
    return feature;
  });

  // 创建矢量数据源
  const vectorSource = new VectorSource({
    features: markers
  });

  // 创建矢量图层
  const vectorLayer = new VectorLayer({
    name: 'customMarkers',
    zIndex: 11,
    source: vectorSource
  });

  // 处理鼠标悬停效果
  let hoveredFeature = null;
  const handlePointerMove = (evt) => {
    const pixel = evt.pixel;
    const map = evt.target;
    
    const hit = map.forEachFeatureAtPixel(pixel, (feature, layer) => {
      if (layer === vectorLayer) {
        return feature;
      }
      return null;
    });

    if (hit !== hoveredFeature) {
      if (hoveredFeature) {
        // 恢复之前悬停要素的样式
        hoveredFeature.setStyle(new Style({
          image: new Icon({
            anchor: [0.5, 0.5],
            src: "data:image/svg+xml," + encodeURIComponent(NetIcons.exchange),
            width: 50
          }),
          text: new Text({
            text: hoveredFeature.get('name'),
            offsetY: -35,
            font: '20px Verdana,sans-serif',
            fill: new Fill({
              color: '#fff'
            })
          })
        }));
      }
      if (hit) {
        // 设置新悬停要素的样式
        hit.setStyle(new Style({
          image: new Icon({
            anchor: [0.5, 0.5],
            src: "data:image/svg+xml," + encodeURIComponent(NetIcons.exchange),
            width: 60  // 悬停时稍微放大
          }),
          text: new Text({
            text: hit.get('name'),
            offsetY: -35,
            font: '20px Verdana,sans-serif',
            fill: new Fill({
              color: '#ffeb3b'  // 悬停时改变颜色
            })
          })
        }));
      }
      hoveredFeature = hit;
    }
  };

  return {
    vectorLayer,
    vectorSource,
    handlePointerMove,
    updateMarker: (index, newCoord) => {
      const feature = markers[index];
      if (feature) {
        feature.getGeometry().setCoordinates(newCoord);
        feature.set('properties', {
          ...feature.get('properties'),
          originalCoord: newCoord
        });
        // 更新后重新设置样式
        feature.setStyle(createStyle(feature, false));
      }
    },
    addMarker: (coord, name) => {
      const feature = new Feature({
        geometry: new Point(coord),
        name: name,
        type: 'icon',
        properties: {
          index: markers.length,
          originalCoord: coord
        }
      });
      feature.setStyle(createStyle(feature, false));
      vectorSource.addFeature(feature);
      markers.push(feature);
    },
    removeMarker: (index) => {
      if (index >= 0 && index < markers.length) {
        vectorSource.removeFeature(markers[index]);
        markers.splice(index, 1);
      }
    }
  };
}

# 方法参数为方法

createClickHandler 是一个用于创建地图点击事件处理器的工厂函数，它封装了地图上不同图层的点击事件处理逻辑，支持多图层点击优先级处理，并提供了弹窗展示功能。

# 技术特点

  1. 工厂函数模式：采用工厂函数模式创建事件处理器，便于复用和配置
  2. 多图层支持：支持处理多个图层的点击事件，包括：
    • 标记点图层（Marker Layer）
    • 基础信息图层（Basic Info Layer）
    • 农场边界图层（Farm Boundary Layer）
  3. 优先级处理：通过 LAYER_PRIORITY 配置实现图层点击优先级
  4. 弹窗管理：支持多个弹窗（Overlay）的管理和切换
  5. 数据更新：支持点击事件触发后的数据更新和展示

# 使用方法

# 1. 基本调用

// 在组件中调用
const clickHandler = createClickHandler(map, {
  overlay,           // 主弹窗覆盖物
  grazingDataArr,    // 数据数组
  globalClearLayer,  // 清除图层函数
  catPannel,         // 面板状态
  getFarmArea,       // 获取农场面积函数
  updatePlantingData // 更新种植数据的方法
});

// 添加点击事件监听
map.on('click', clickHandler);

# 2. 参数说明

/**
 * @param {Map} map - OpenLayers地图实例
 * @param {Object} options - 配置选项
 * @param {Overlay} options.overlay - 主弹窗覆盖物
 * @param {Ref} options.grazingDataArr - 数据数组（响应式引用）
 * @param {Function} options.globalClearLayer - 清除图层函数
 * @param {Ref} options.catPannel - 面板状态（响应式引用）
 * @param {Function} options.getFarmArea - 获取农场面积函数
 * @param {Function} options.updatePlantingData - 更新种植数据的方法
 * @returns {Function} 返回点击事件处理函数
 */

# 3. 图层优先级配置

export const LAYER_PRIORITY = [
  LAYER_NAMES.MARKER,        // 标记点图层优先级最高
  LAYER_NAMES.BASIC_INFO,    // 基础信息图层次之
  LAYER_NAMES.FARM_BOUNDARY  // 农场边界图层优先级最低
];

# 4. 事件处理流程

  1. 点击事件触发：

    • 获取点击位置的像素坐标
    • 收集点击位置的所有要素

  2. 优先级处理：

    • 根据 LAYER_PRIORITY 对点击的要素进行排序
    • 处理优先级最高的图层点击事件

  3. 图层特定处理：

    • 标记点图层：显示标记点信息
    • 基础信息图层：显示农场信息和种植数据
    • 农场边界图层：显示农场边界信息

  4. 弹窗管理：

    • 根据图层类型显示对应的弹窗
    • 处理弹窗的显示和隐藏
    • 更新弹窗内容
