// 绘制曲线
const drawCurve = (curve, progress) => {
  if (!ctx.value) return;

  const { start, control1, control2, end } = curve.points;
  const { lineStyle } = curve;
  const width = canvasRef.value.width;
  const height = canvasRef.value.height;

  // 计算实际坐标
  const getPoint = (point) => ({
    x: point.x * width,
    y: point.y * height
  });

  const startPoint = getPoint(start);
  const controlPoint1 = getPoint(control1);
  const controlPoint2 = getPoint(control2);
  const endPoint = getPoint(end);

  // 创建渐变
  const gradient = ctx.value.createLinearGradient(
    startPoint.x, 0, endPoint.x, 0
  );
  gradient.addColorStop(0, curve.color + '99');
  gradient.addColorStop(1, curve.color);

  // 设置样式
  ctx.value.strokeStyle = gradient;
  ctx.value.lineWidth = lineStyle.lineWidth;
  ctx.value.lineCap = 'round';
  ctx.value.lineJoin = 'round';

  // 开始绘制路径
  ctx.value.beginPath();
  ctx.value.moveTo(startPoint.x, startPoint.y);

  // 使用三阶贝塞尔曲线
  ctx.value.bezierCurveTo(
    controlPoint1.x, controlPoint1.y,
    controlPoint2.x, controlPoint2.y,
    endPoint.x, endPoint.y
  );

  // 添加阴影
  ctx.value.shadowColor = 'rgba(0, 0, 0, 0.2)';
  ctx.value.shadowBlur = 4;
  ctx.value.shadowOffsetY = 2;

  // 根据配置设置虚线样式
  if (lineStyle.isDashed) {
    ctx.value.setLineDash([lineStyle.dashLength, lineStyle.gapLength]);
    ctx.value.lineDashOffset = -progress * (lineStyle.dashLength + lineStyle.gapLength) * 2;
  } else {
    ctx.value.setLineDash([]); // 实线
  }

  ctx.value.stroke();

  // 重置阴影和虚线样式
  ctx.value.shadowColor = 'transparent';
  ctx.value.setLineDash([]);
};

// 修改动画循环函数
const animate = () => {
  if (!ctx.value || !canvasRef.value) return;

  // 清除画布
  ctx.value.clearRect(0, 0, canvasRef.value.width, canvasRef.value.height);

  // 更新和绘制每条曲线
  curves.value.forEach(curve => {
    if (curve.visible) {
      // 更新进度
      if (curve.progress < 1) {
        curve.progress += 0.005; // 减慢动画速度
      } else {
        curve.progress = 0; // 重置进度，实现循环动画
      }
      drawCurve(curve, curve.progress);
    }
  });

  // 继续动画循环
  animationFrameId.value = requestAnimationFrame(animate);
};

// 开始动画
const startAnimation = () => {
  if (animationFrameId.value) {

  animate();
};

const updateNetworkStatus = () => {
  networkStatus.value = networkStatus.value.map((item) => {
    const random = Math.random();

    if (random < 0.2) {
      return {
        status: "error",
        statusText: "异常",
        delay: Math.floor(Math.random() * 200) + 100,
        packetLoss: (Math.random() * 10 + 5).toFixed(1),
      };
    } else if (random < 0.5) {
      return {
        status: "warning",
        statusText: "注意",
        delay: Math.floor(Math.random() * 50) + 40,
        packetLoss: (Math.random() * 3 + 1).toFixed(1),
      };
    } else {
      return {
        status: "normal",
        statusText: "正常",
        delay: Math.floor(Math.random() * 30) + 10,
        packetLoss: (Math.random() * 0.2).toFixed(2),
      };
    }
  });
};
