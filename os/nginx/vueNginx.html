# 1. createWebHistory 配置详解

# 1.1 不带参数 createWebHistory()

const router = createRouter({
  history: createWebHistory(),
  routes
})

访问方式：

  • 根路径访问：http://your-domain.com/
  • 子路由访问：http://your-domain.com/topic1Video
  • 带参数访问：http://your-domain.com/topic1Video?id=123

部署影响：

  • 应用部署在网站根目录
  • 所有路由都是相对于根路径
  • 静态资源路径：http://your-domain.com/assets/xxx.js

后端接口配置：

# Nginx 配置
server {
    listen 80;
    server_name your-domain.com;

    # 前端应用
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
    }

    # 后端 API 代理
    location /api/ {
        proxy_pass http://localhost:5123/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

# 1.2 带参数 createWebHistory('/my-demo/')

const router = createRouter({
  history: createWebHistory('/my-demo/'),
  routes
})

访问方式：

  • 根路径访问：http://your-domain.com/my-demo/
  • 子路由访问：http://your-domain.com/my-demo/topic1Video
  • 带参数访问：http://your-domain.com/my-demo/topic1Video?id=123

部署影响：

  • 应用部署在子目录
  • 所有路由都是相对于 /my-demo/ 路径
  • 静态资源路径：http://your-domain.com/my-demo/assets/xxx.js

后端接口配置：

server {

    server_name your-domain.com;

    # 前端应用
    location /my-demo/ {
        alias /usr/share/nginx/html/;
        try_files $uri $uri/ /my-demo/index.html;
    }

    # 后端 API 代理
    location /my-demo/api/ {
        proxy_pass http://localhost:5123/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

# 3. Nginx 配置详解

# 3.1 根路径配置

server {
    listen 80;
    server_name your-domain.com;

    # 前端应用
    location / {
        root /usr/share/nginx/html;
        index index.html;
        try_files $uri $uri/ /index.html;

        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
            expires 1y;
            add_header Cache-Control "public, no-transform";

    location /api/ {
        proxy_pass http://localhost:5123/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # API 缓存控制
        add_header Cache-Control "no-cache, no-store, must-revalidate";

    add_header X-Frame-Options "SAMEORIGIN";
    add_header X-XSS-Protection "1; mode=block";

# 关于 vite.config.js base配置和 router 路由参数

是的，当 vite.config.js 配置了 base: '/bjtu/' 时，路由也必须相应配置 createWebHistory('/bjtu/')，这是因为：

  1. Vite 的 base 配置：

    • 影响所有静态资源的加载路径
    • 影响开发服务器的基础路径
    • 影响构建后的资源引用路径

  2. Vue Router 的 createWebHistory：

    • 控制前端路由的基础路径
    • 影响路由的匹配和跳转
    • 影响浏览器历史记录的管理

当前你的配置：

// vite.config.js
export default defineConfig({
  base: '/bjtu/',  // 已配置
  // ... 其他配置
})

// router/index.js
const router = createRouter({
  history: createWebHistory(),  // 未配置，使用默认值 '/'
  routes
})

这样配置会导致以下问题：

  1. 路径不匹配：

    • 静态资源路径：/bjtu/assets/xxx.js（由 Vite 的 base 决定）
    • 路由路径：/topic1Video（由 Router 的默认值决定）
    • 导致路由无法正确匹配到资源

  2. 访问不到页面的原因：

    • 浏览器访问：http://your-domain.com/bjtu/topic1Video
    • 但路由期望：http://your-domain.com/topic1Video
    • 造成路径不匹配，导致 404

正确的配置应该是：

// vite.config.js
export default defineConfig({
  base: '/bjtu/',
  // ... 其他配置
})

// router/index.js
const router = createRouter({
  history: createWebHistory('/bjtu/'),  // 与 vite.config.js 的 base 保持一致
  routes
})

# nginx.conf 斜杠的问题

我来详细解释 Nginx 中 location 的 root 和 alias 的区别，以及斜杠的影响：

# 1. root 和 alias 的区别

# 1.1 root 指令

location /bjtu/ {
    root /usr/share/nginx/html;  # 不带斜杠
    # 或
    root /usr/share/nginx/html/; # 带斜杠
}

特点：

  • root 会将 location 的路径追加到 root 路径后面
  • 最终路径 = root路径 + location路径 + 请求路径
  • 例如访问 /bjtu/index.html：
    • 实际文件路径：/usr/share/nginx/html/bjtu/index.html

# 1.2 alias 指令

location /bjtu/ {
    alias /usr/share/nginx/html;  # 不带斜杠
    # 或
    alias /usr/share/nginx/html/; # 带斜杠
}

特点：

  • alias 会用 alias 路径替换 location 路径
  • 最终路径 = alias路径 + 请求路径（不包含 location 路径）
  • 例如访问 /bjtu/index.html：
    • 实际文件路径：/usr/share/nginx/html/index.html

# 2. 斜杠的影响

# 2.1 不带斜杠的情况

# 配置1：都不带斜杠
location /bjtu {
    root /usr/share/nginx/html;
    # 访问 /bjtu/index.html
    # 实际路径：/usr/share/nginx/html/bjtu/index.html
}

location /bjtu {
    alias /usr/share/nginx/html;
    # 访问 /bjtu/index.html
    # 实际路径：/usr/share/nginx/html/index.html
}

# 配置2：location带斜杠，路径不带斜杠
location /bjtu/ {
    root /usr/share/nginx/html;
    # 访问 /bjtu/index.html
    # 实际路径：/usr/share/nginx/html/bjtu/index.html
}

location /bjtu/ {
    alias /usr/share/nginx/html;
    # 访问 /bjtu/index.html
    # 实际路径：/usr/share/nginx/html/index.html
}

# 2.2 带斜杠的情况

# 配置3：都带斜杠
location /bjtu/ {
    root /usr/share/nginx/html/;
    # 访问 /bjtu/index.html
    # 实际路径：/usr/share/nginx/html/bjtu/index.html
}

location /bjtu/ {
    alias /usr/share/nginx/html/;
    # 访问 /bjtu/index.html
    # 实际路径：/usr/share/nginx/html/index.html
}

# 3. 实际应用示例

# 3.1 正确的配置（推荐）

server {
    listen 80;
    server_name your-domain.com;

    # 方式1：使用 alias（推荐）
    location /bjtu/ {
        alias /usr/share/nginx/html/;  # 带斜杠
        try_files $uri $uri/ /bjtu/index.html;
    }

    # 方式2：使用 root
    location /bjtu/ {
        root /usr/share/nginx/html;    # 不带斜杠
        try_files $uri $uri/ /bjtu/index.html;
    }

    # 静态资源缓存
    location /bjtu/assets/ {
        alias /usr/share/nginx/html/assets/;  # 带斜杠
        expires 1y;
        add_header Cache-Control "public, no-transform";
    }

    # API 代理
    location /bjtu/api/ {
        proxy_pass http://localhost:5123/;  # 带斜杠
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

# 3.2 错误的配置（会导致问题）

# 错误1：location 不带斜杠
location /bjtu {  # 错误
    alias /usr/share/nginx/html/;
    # 访问 /bjtu/index.html 会变成 /usr/share/nginx/html/bjtu/index.html
}

# 错误2：alias 路径不带斜杠
location /bjtu/ {
    alias /usr/share/nginx/html;  # 错误
    # 访问 /bjtu/index.html 会变成 /usr/share/nginx/htmlindex.html
}

# 错误3：root 和 location 都不带斜杠
location /bjtu {  # 错误
    root /usr/share/nginx/html;  # 错误
    # 访问 /bjtu/index.html 会变成 /usr/share/nginx/html/bjtu/index.html
}

# 4. 最佳实践建议

  1. 使用 alias 的情况：

    • 当 location 路径需要完全替换时
    • 部署在子路径时
    • 需要精确控制文件路径时

  2. 使用 root 的情况：

    • 当 location 路径需要保留时
    • 部署在根路径时
    • 需要保持目录结构时

  3. 斜杠使用建议：

    • location 路径：总是带斜杠 /bjtu/
    • alias 路径：总是带斜杠 /usr/share/nginx/html/
    • root 路径：可以不带斜杠 /usr/share/nginx/html

  4. 完整的推荐配置：

server {
    listen 80;
    server_name your-domain.com;

    # 前端应用
    location /bjtu/ {
        alias /usr/share/nginx/html/;
        try_files $uri $uri/ /bjtu/index.html;
        
        # 安全头
        add_header X-Frame-Options "SAMEORIGIN";
        add_header X-XSS-Protection "1; mode=block";
        add_header X-Content-Type-Options "nosniff";
    }

    # 静态资源
    location /bjtu/assets/ {
        alias /usr/share/nginx/html/assets/;
        expires 1y;
        add_header Cache-Control "public, no-transform";
    }

    # API 代理
    location /bjtu/api/ {
        proxy_pass http://localhost:5123/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

这样配置可以确保：

  1. 路径正确匹配
  2. 文件正确访问
  3. 路由正常工作
  4. 静态资源正确加载
  5. API 正确代理
  6. 缓存策略合理
  7. 安全性得到保障

# ./ 的情况

# 1. base: './' 的含义

// vite.config.js
export default defineConfig({
  base: './',  // 使用相对路径
  // ... 其他配置
})

这种配置的特点：

  1. 使用相对路径加载资源
  2. 所有资源路径都会变成相对于当前 HTML 文件的路径
  3. 适合部署在任意子目录，不需要指定具体路径

# 2. 不同 base 配置的对比

// 配置1：绝对路径
base: '/bjtu/'  
// 资源路径：/bjtu/assets/xxx.js
// 需要 Nginx 配置 location /bjtu/

// 配置2：相对路径
base: './'  
// 资源路径：./assets/xxx.js
// 可以部署在任何目录

// 配置3：根路径
base: '/'  
// 资源路径：/assets/xxx.js
// 需要部署在根目录

# 3. 使用 base: './' 的场景

  1. 静态文件部署：

    • 直接打开 HTML 文件
    • 部署在本地文件系统
    • 不需要服务器

  2. 任意子目录部署：

    • 可以部署在任何目录
    • 不需要修改 Nginx 配置
    • 便于迁移和部署

  3. CDN 部署：

    • 资源可以部署在不同域名
    • 支持跨域资源加载
    • 便于资源分发

# 5. 优缺点分析

优点：

  1. 部署灵活，可以放在任何目录
  2. 不需要修改服务器配置
  3. 支持直接打开 HTML 文件
  4. 便于本地开发和测试
  5. 适合静态文件部署

缺点：

  1. 可能影响缓存策略
  2. 某些 CDN 可能不支持相对路径
  3. 需要确保所有资源路径正确
  4. 可能影响 SEO
  5. 某些浏览器安全策略可能限制

# 6. 使用建议

  1. 适合使用 base: './' 的场景：

    • 本地开发测试
    • 静态文件部署
    • 需要频繁迁移部署
    • 不需要固定部署路径

  2. 不适合使用 base: './' 的场景：

    • 需要固定部署路径
    • 使用 CDN 加速
    • 需要严格的缓存控制
    • 需要 SEO 优化

  3. 最佳实践：

// vite.config.js
export default defineConfig({
  base: process.env.NODE_ENV === 'production' 
    ? './'  // 生产环境使用相对路径
    : '/',  // 开发环境使用绝对路径
  // ... 其他配置
})

// router/index.js
const router = createRouter({
  history: createWebHistory(process.env.NODE_ENV === 'production' ? './' : '/'),
  routes
})
  1. 注意事项：
    • 确保所有资源路径正确
    • 测试不同环境下的访问
    • 考虑缓存策略
    • 注意浏览器安全策略
    • 考虑 SEO 影响

总的来说，base: './' 是一个灵活的配置，适合需要部署灵活性的场景，但也要考虑其优缺点和适用场景。

# 两种不一样的方式部署和配置

我来详细解释这两种配置的区别和对应的配置方式：

# 1. base: '/bjtu/' 配置

# 1.1 Vite 配置

// vite.config.js
export default defineConfig({
  base: '/bjtu/',  // 绝对路径
  // ... 其他配置
})

# 1.2 路由配置

// router/index.js
const router = createRouter({
  history: createWebHistory('/bjtu/'),  // 必须匹配 base 配置
  routes
})

# 1.3 Nginx 配置

server {
    listen 80;
    server_name your-domain.com;

    # 方式1：使用 alias（推荐）
    location /bjtu/ {
        alias /usr/share/nginx/html/;
        try_files $uri $uri/ /bjtu/index.html;
    }

    # 方式2：使用 root
    location /bjtu/ {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /bjtu/index.html;
    }
}

# 1.4 访问方式

  • 首页：http://your-domain.com/bjtu/
  • 子页面：http://your-domain.com/bjtu/topic1Video
  • 静态资源：http://your-domain.com/bjtu/assets/xxx.js

# 1.5 原理

  1. 绝对路径配置：

    • 所有资源路径都从域名根目录开始
    • 需要 Nginx 配置对应的 location
    • 路由必须匹配这个基础路径
    • 资源加载路径固定

  2. 部署特点：

    • 必须部署在指定路径
    • 需要服务器配置支持
    • 路径固定，便于管理
    • 适合生产环境

# 2. base: './bjtu/' 配置

# 2.1 Vite 配置

// vite.config.js
export default defineConfig({
  base: './bjtu/',  // 相对路径
  // ... 其他配置
})

# 2.2 路由配置

// router/index.js
const router = createRouter({
  history: createWebHistory('./bjtu/'),  // 必须匹配 base 配置
  routes
})

# 2.3 Nginx 配置

server {
    listen 80;
    server_name your-domain.com;

    # 方式1：部署在根目录
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
    }

    # 方式2：部署在任意子目录
    location /any/path/ {
        alias /usr/share/nginx/html/;
        try_files $uri $uri/ /any/path/index.html;
    }
}

# 2.4 访问方式

  • 直接打开：file:///path/to/dist/bjtu/index.html
  • 本地服务器：http://localhost:8080/bjtu/index.html
  • 任意子目录：http://your-domain.com/any/path/bjtu/index.html

# 2.5 原理

  1. 相对路径配置：

    • 资源路径相对于 HTML 文件位置
    • 不依赖服务器配置
    • 可以部署在任何目录
    • 资源加载路径动态

  2. 部署特点：

    • 可以部署在任意目录
    • 不需要特定服务器配置
    • 路径灵活，便于迁移
    • 适合开发环境

# 3. 两种配置的对比

# 3.1 路径处理

# 4. 最佳实践建议

  1. 开发环境配置：
// vite.config.js
export default defineConfig({
  base: process.env.NODE_ENV === 'development' 
    ? './bjtu/'  // 开发环境使用相对路径
    : '/bjtu/',  // 生产环境使用绝对路径

    process.env.NODE_ENV === 'development' 
      ? './bjtu/'

  1. 生产环境 Nginx 配置：
server {
    listen 80;
    server_name your-domain.com;

    # 使用绝对路径配置
    location /bjtu/ {
        alias /usr/share/nginx/html/;
        try_files $uri $uri/ /bjtu/index.html;
        
        # 缓存控制
        location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
            expires 1y;
            add_header Cache-Control "public, no-transform";
        }
    }

    # API 代理
    location /bjtu/api/ {
        proxy_pass http://localhost:5123/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}
  1. 构建和部署脚本：
#!/bin/bash

# 构建
npm run build

# 生产环境部署（绝对路径）
if [ "$ENV" = "production" ]; then
    cp -r dist/* /usr/share/nginx/html/
    # 确保 bjtu 目录存在
    mkdir -p /usr/share/nginx/html/bjtu
    # 移动文件到 bjtu 目录
    mv /usr/share/nginx/html/index.html /usr/share/nginx/html/bjtu/
    mv /usr/share/nginx/html/assets /usr/share/nginx/html/bjtu/
fi

# 开发环境部署（相对路径）
if [ "$ENV" = "development" ]; then
    cp -r dist/* /usr/share/nginx/html/bjtu/
fi

    • 确保路径配置一致性
    • 测试不同环境下的访问
    • 考虑缓存策略
    • 注意资源加载顺序
    • 考虑浏览器兼容性
