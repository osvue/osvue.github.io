# logback的配置⽂件必须放在项⽬根⽬录中 且名字必须为 logback.xml

<?xml version="1.0" encoding="UTF-8" ?>
<configuration>
 <!--定义项⽬中⽇志输出位置-->
 <appender name="stdout"
class="ch.qos.logback.core.ConsoleAppender">
 <!--定义项⽬的⽇志输出格式-->
 <!--定义项⽬的⽇志输出格式-->
 <layout
class="ch.qos.logback.classic.PatternLayout">
 <pattern> [%p] %d{yyyy-MM-dd HH:mm:ss}
%m %n</pattern>
 </layout>
 </appender>
 <!--项⽬中跟⽇志控制-->
 <root level="INFO">
 <appender-ref ref="stdout"/>
 </root>
 <!--项⽬中指定包⽇志控制-->
 <logger name="com.osvue.lcon" level="DEBUG"/>
</configuration>
