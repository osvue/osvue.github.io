(function controlbtn(){
	//$("[uri]").css('visibility','hidden');//元素隐藏
	var arr = [];
	var page_type = true;//默认开启类型验证
	if(typeof cur_type === 'undefined') page_type = false;//没有属性,则关闭类型
	$("[uri]").each(function(i){
		if(page_type && cur_type != $(this).attr("flag")){
			$(this).remove();
		}else{
			arr.push({id:$(this).attr("id"),uri:$(this).attr("uri")})
		}
	})
	$security.ajax({
		url : "请求资源判断是否合法访问",
		data : {
			arr : JSON.stringify(arr)
		},
		type : 'post',
		dataType : 'json',
		success : function(data) {
		//  data 为 合法访问数据 ， 遍历页面数据 ，并隐藏未授权数据
			for(var i=0;i<arr.length;i++){

				if(data.indexOf(arr[i].id)==-1){
                    // 特殊处理 的一些 tabs 标签
                    if(arr[i].id.indexOf("home") > -1){
					 $("#jcglTabs").tabs("close", "tables_tab");
                    } else{
                        // 代办里的按钮
                        $("#"+arr[i].id).remove();
                    }
					   
				} else {
						// 如果是代办环节页签
						$("#"+arr[i].id).show();
 				}
			}
		}
	});
})()
